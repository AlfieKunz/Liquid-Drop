	Force per unit area on fluid element: $\underline{F} = \int p \underline{n}\cdot d\underline{S} = \int_\Omega\nabla \cdot \mathbf{T} \ dV = \int_A \mathbf{T}\cdot \underline{n}\ dA$, where $\mathbf{T}\cdot \underline{n}$ is the stress on a surface normal to $\underline{n}$.
	
	$T_{ij} = -p\delta_{ij} + 2\mu e_{ij}$ is the Stress Tensor. Components $\{i,j\}$ is the stress along $i$ on the surface normal to $j$.
		
	Material Derivative: ${\frac{D}{Dt}} := {\frac{\partial }{\partial t} + (\underline{u}\cdot \nabla )}$.

	Navier Stokes Equations:
	\begin{subequations}
		\label{NavierStokes}
		\begin{align}
			\nabla \cdot \underline{u} = 0,\\
			\underbrace{\rho\left( \frac{\partial \underline{u}}{\partial t} + \overbrace{(\underline{u}\cdot \nabla) \underline{u}}^\text{inertial} \right)}_{ma} = \underbrace{\overbrace{\mu\nabla ^2 \underline{u}}^\text{viscous} - \nabla p + \rho \underline{g}}_{\sum \underline{F}}.
		\end{align}
	\end{subequations}
	
	Stokes' Equations:
	\begin{subequations}
		\label{Stokes}
		\begin{align}
			\nabla p = \mu \nabla ^2 \underline{u},\\
			\nabla \cdot \underline{u} = 0.
		\end{align}
	\end{subequations}
	
	Streamlines: $\frac{\partial \underline{x}(s)}{\partial s} = \underline{u}(\underline{x}(s),t)$.
	
	Reynolds Number: $\text{Re} = \frac{\rho UR}{\mu}$.
	
	Bernoulli's Equations:
	\begin{subequations}\label{Bernoulli}
		\begin{align}
			\tilde{B}(\underline{x}) &= \left( \frac{\rho}{2} |u(x)|^2 + p  + \rho gz \right),\quad&\text{(Steady, Inviscid. Constant on streamlines)} \label{Bernoulli1}\\
			\hat{B}(t) &= \left( {\rho \frac{\partial \phi}{\partial t}} + \frac{\rho}{2} |u(x,t)|^2 + p + \rho gz \right).\quad&\text{(Potential. Constant in space)} \label{Bernoulli2}
		\end{align}
	\end{subequations}
	
	Potential / Irrotational Flows: $\underline{u} = \nabla \phi(\underline{x},t) = \left( \frac{\partial \phi}{\partial r}, \frac{1}{r}\frac{\partial \phi}{\partial \theta} \right)$, with $\nabla ^2\phi = 0$.

	\section{Problem Overview}
	Bounces are modelled specifically via the thin air film that coats the drop - lubrication theory. The drop bounces on it's \textit{own air cushion}.
	Collisions with solids don't happen unless the solid is nano-smooth - small ridges puncture air film and reaches the liquid, sticking to it. Hence, drop-drop collisions (or symmetric to halve computation?).
	
	The closer the drop gets to the barrier (and the thinner the air film gets), the more pressure forms. With no attractive forces, this pressure builds up until it repels the drop. At this initial phase, we are attempting to recreate figure 3a from Hicks and Pervis.
	
	
	The drop is able to reach the barrier for two reasons: at small scales, the attract each other through van der Waals forces, and if the film height drops below the mean free path of gas molecules, pressure decreases, causing the gas molecules to slip along the boundary (Knudsen Gas-Kinetic Slip). If this happens, the the film ruptures and the drops merge.

	\subsection{Recap of Plane-Poiseuille Flow, and Lubrication Theory}
	The Plane-Poiseuille flow is the flow between two stationary plates of separation $h \sim \hat{\underline{z}}$.
	\begin{itemize}
		\item By the profile, we write $\underline{u} = u(y)\underline{\hat{x}}$. Via 1D nature, \textcolor[RGB]{255, 170, 0}{conservation of momentum} gives \underline{no inertial terms}, and:
		\begin{itemize}
			\item $\color{orange}x$: $\color{cyan}0 = -\frac{\partial p}{\partial x} + \mu \frac{\partial ^2u}{\partial y^2}$ \textit{(friction moves from boundary into the fluid)}, \quad $\color{orange}y$: $\color{lime} 0= -\frac{\partial p}{\partial y}$.
		\end{itemize}
		\item As $\color{lime}p(x)$, $u(y)$, we just \textcolor[RGB]{85, 255, 255}{integrate} to get $u(y) = \left( \frac{1}{2\mu} \frac{d p}{d x} \right) y^2 + Ay + B$.
		\item Using $u(y = \pm h/2) = 0$ (\underline{no-slip}, viscous), we get the \textbf{parabola} $\underline{u(y) = \frac{1}{2\mu} \frac{d p}{d x}\left( y^2 - \frac{h^2}{4} \right)}$.
	\end{itemize}
	
	\textbf{Lubrication Theory}: used as film radius $r \gg h$ height - very weak parabola. Hence, a fine assumption in this simple case is that the Poiseuille profile holds: $x,x^2 \to 0$ in the gas film, and we get a strip.

	\subsection{Lubrication Equation Derivation}
	By the axisymmetric conditions, $\partial / \partial \theta = 0$ and $u_\theta = 0$ \todo[color=cyan]{From "Stability of nonswirl axisymmetric solutions to the Navier-Stokes equations"}, and thus we get $\underline{u}(r,z)$. Hence, in the absence of external forces (including gravity), we get:
	
	\begin{subequations}\label{NavierStokesAXI}
		\begin{align}
			\frac{1}{r} \frac{\partial (r u_r)}{\partial r} + \frac{\partial u_z}{\partial z} &= 0, \label{NavierStokesAXI:continuity} \\
			\rho_g \left( \frac{\partial u_r}{\partial t} + u_r \frac{\partial u_r}{\partial r} + u_z \frac{\partial u_r}{\partial z} \right) &= -\frac{\partial p_g}{\partial r} + \mu_g \left[ \frac{\partial}{\partial r}\left( \frac{1}{r} \frac{\partial (r u_r)}{\partial r} \right) + \frac{\partial^2 u_r}{\partial z^2} \right], \label{NavierStokesAXI:r} \\
			\rho_g \left( \frac{\partial u_z}{\partial t} + u_r \frac{\partial u_z}{\partial r} + u_z \frac{\partial u_z}{\partial z} \right) &= -\frac{\partial p_g}{\partial z} + \mu_g \left[ \frac{1}{r}\frac{\partial}{\partial r}\left( r \frac{\partial u_z}{\partial r} \right) + \frac{\partial^2 u_z}{\partial z^2} \right]. \label{NavierStokesAXI:z}
		\end{align}
	\end{subequations}
	
	Rescaling the momentum equations under the regime of $h / R \ll 1$ (ie: locally flat), we conclude that, to leading order\todo[color=cyan]{cite Schlichting, 1968, Long-scale evolution of thin liquid films}:
	\begin{subequations}\label{BoundaryLayerScaling}
		\begin{align}
			\frac{\partial p_g}{\partial r} = \mu_g \frac{\partial^2 u_r}{\partial z^2},\label{BLScaling} \\
			\frac{\partial p_g}{\partial z} = 0.
		\end{align}
	\end{subequations}
	This directly concludes that $p_g$ is a function of $r$ and $t$ only! Using the boundary conditions of $\underline{u}(r,0) = \underline{u}(r,h(r,t)) = 0$, we can integrate \eqref{BLScaling} to get the Plane Poiseuille velocity profile \todo[color=cyan]{CITE: Long-scale evolution of thin liquid films}:
	\begin{equation}
		u_r(r,z,t) = \frac{z(z-h(r,t))}{2\mu_g} \frac{\partial p_g(r,t)}{\partial r},
	\end{equation}
	with velocity flux,
	\begin{equation}
		\label{Flux}
		Q(r,t) = \int^{h(r,t)}_0 u_r\, dz = -\frac{h^3}{12\mu_g} \frac{\partial p_g(r,t)}{\partial r}.
	\end{equation}
	To determine the evolution of the film thickness $h(r,t)$, we integrate the continuity equation \eqref{NavierStokesAXI:continuity} inside the entire $z$ domain \todo[color=cyan]{CITE NOTES 2 DERIVATION OF THE CLASSICAL REYNOLDS EQUATION FOR THIN FILM FLOWS} to get,
	\begin{equation}
		\label{ContInt}
		\int_0^{h(r,t)} \frac{1}{r} \frac{\partial (r u_r)}{\partial r}\ dz + \int_0^{h(r,t)} \frac{\partial u_z}{\partial z} \ dz = 0.
	\end{equation}
	Recalling Leibniz Integration Rule, and applying \eqref{BCKinetic}\todo[color=cyan]{CITE: BJ Hamrock, Fundamentals of Fluid Film Lubrication}, the first term in \eqref{ContInt} evaluates to
	\begin{equation}
		\label{ContInt1}
		\frac{1}{r} \frac{\partial}{\partial r}\left( r \int_0^h u_r\, dz \right) - u_r(r,h)\frac{\partial h}{\partial r} \ =\ \frac{1}{r} \frac{\partial (r Q)}{\partial r} - \left( u_z(r,h) - \frac{\partial h}{\partial t} \right),
	\end{equation}
	whilst the Fundamental Theorem of Calculus and boundary condition \eqref{BCNoSlip} can be used to evaluate the second term in \eqref{ContInt} as merely $u_z(r,h)$. Combining this with \eqref{ContInt1} in continuity equation \eqref{ContInt}, whilst subbing in \eqref{Flux} for $Q$, we retrieve the axisymmetric Reynolds lubrication equation \todo[color=cyan]{CITE: Modelling and closed loop control of near-field acoustically levitated objects, A rotational traveling wave based levitation device - Modelling, design, and control }:
	\begin{equation}
		\frac{\partial h(r,t)}{\partial t} = \frac{1}{r}\frac{\partial }{\partial r}\left( \frac{rh^3}{12\mu_g} \frac{\partial p_g(r,t)}{\partial r} \right).
	\end{equation}
	Scaling radial lengths by the drop radius $R$, film thickness by $H$, time by $R/U$, and pressures by the liquid inertial scale $\rho_l U^2$, this can be expressed in non-dimensional form as:
	\begin{equation}\label{LubricationND}
		\frac{\partial h(r,t)}{\partial t} = \frac{1}{r}\frac{\partial }{\partial r}\left( \frac{\text{St}}{12} rh^3 \frac{\partial p_g(r,t)}{\partial r} \right),
	\end{equation}
	where $\text{St} = \frac{\rho_l R U}{\mu_g}$ is the Stokes number.
	
	\subsection{Droplet Equations Formation}
	We must reset our assumptions of a thin film, and model $p_l(r,z)$ specifically.
	We introduce the characteristic pressure length $\ell$ (the horizontal length that pressure makes a leading order contribution to the deformation of the drop), or rather the ratio $\delta := \ell/R$. Rescaling with this ($L \to \delta L$, $t \to \delta^2t$, $h \to \delta^2h$, $U \to U$ (and thus other components of drop velocity! Don't really care about oblique impacts unless they are VERY oblique - solutions match to leading order), $p \to p/\delta$), to leading-order as $z \to 0$ we lose the advective terms fo the material derivative $\frac{D}{Dt}$ \todo[color=cyan]{CITE: HP, Air cushioning with a lubrication/inviscid balance}. This has the following consequences on our formulation:
	\begin{itemize}
		\item \eqref{BCKinetic} becomes\todo[color=cyan]{CITE HP}
		\begin{equation}
			\label{KineticApprox}
			\frac{\partial h}{\partial t} = u_z(r, h).
		\end{equation}
		In other words, the interface $f(r,z)$ flattens entirely onto the $z=0$ plane.
		\item Recalling that we are in a high Reynolds Number approximation, the axisymmetric version of \eqref{NavierStokesND} becomes:
		\begin{equation}
			\label{NSHarmonic}
			\frac{\partial \underline{u}_l}{\partial t} = -\left( \frac{\partial p_l}{\partial r}, \frac{\partial p_l}{\partial z} \right).
		\end{equation}
	\end{itemize}
	(Again, good to stress that this is in the leading order approximation of $z \to 0$.)
	
	Considering initially the second of these points, taking the cylindrical divergence of \eqref{NSHarmonic}, and equating to \eqref{NavierStokesAXI:continuity} gives \todo[color=cyan]{CITE: HP, Air cushioning with a lubrication/inviscid balance},
	\begin{equation}
		\label{PressureLaplace}
		\nabla ^2 p_l = \frac{1}{r}\frac{\partial}{\partial r}\left(r \frac{\partial p_l}{\partial r}\right) + \frac{\partial^2 p_l}{\partial z^2} = 0.
	\end{equation}
	In other words, the pressure field is Harmonic within the liquid. By the weak maximum principle for the Laplace Equation, we can therefore determine $p$ uniquely from the boundary conditions (for example, using the Dirichlet Principle). \todo[color=cyan]{cite: Evans, L. C. (2010). Partial Differential Equations}

	Considering now the first of these points, we are able to relate the downward acceleration of the interface to the pressure at the boundary by differentiating \eqref{KineticApprox} and using \eqref{NSHarmonic} to get:
	\begin{equation}
		\label{KineticODE}
		-\frac{\partial^2 h(r,t)}{\partial t^2} = \left.\frac{\partial p_l(r,z)}{\partial z}\right|_{z \to 0^+},
	\end{equation}		
	providing an ODE for the interface, which can be solved by the following recipe:
	\begin{itemize}
		\item Solve the Lubrication Equation for the thin film, given the previously produced value (or initial condition) for $\partial_t h(r,t)$, giving us an equation for $p_g(r,t)$.
		\item Use Bernoulli's equations to solve for the pressure boundary conditions of the liquid, via $p_l(r, h) = p_g(r, t)$ (exploiting the fact that $\partial _zp_g = 0$).
		\item Exploit the linearity of the harmonic liquid pressure to find an equation for the full liquid pressure.
		\item Use this pressure equation to thus solve the ODE \eqref{KineticODE}, giving us a new value for $h(r,t)$.
		\item Repeat!
	\end{itemize}

	\begin{equation}
		\label{PressureDelta}
		p_l(r,z) = \iint \delta(r-r', z-z')p_l(r',z') 2\pi r' \ dr'dz' .
	\end{equation}

		\begin{align}
			\nabla &:= \left( \frac{\partial }{\partial r'}\ ,\ \frac{\partial }{\partial z} \right), \label{NablaCyl:1} \\
			\nabla^2 &:= \frac{1}{r'} \frac{\partial }{\partial r'} \left( r' \frac{\partial }{\partial r'} \right) + \frac{\partial ^2}{\partial z^2}. \label{NablaCyl:2}
		\end{align}
	\end{subequations}
	
	Recalling that the eigenfunctions of the radial components of \eqref{NablaCyl:2} are the 0th-order Bessel Functions $J_0(kr')$ (where $k$ are eigenvector wavenumbers), which satisfy $\frac{1}{r'}\frac{d}{dr'}\big(r'\frac{d}{dr'}J_0(k r')\big) = -k^2 J_0(k r')$, we can write \eqref{GreenFreeSpace} in pure $(r,z)$ components by expressing it as a Hankel integral:
	\begin{equation}
		\label{GreenHankel}
		G(r, z; r', z') = \frac{1}{4\pi} \int_0^\infty J_0(k r) J_0(k r') e^{-k|z-z'|} \, dk.
	\end{equation}
	Lipschitz–Hankel proved that this has an exact closed-form solution:
	\begin{equation}
		\label{GreenElliptic}
		G(r,z; r',z') = \frac{K(m_{\_}),m_{\_}}{2\pi^2\sqrt{(r+r')^2+(z-z')^2}} = \frac{4rr'}{(r+r')^2 + (z-z')^2},
	\end{equation}
	where $K(m) = \int^{\pi/2}_0 \frac{d\theta}{\sqrt{1-m\sin^2(\theta)}}$ is the complete elliptical integral of the first kind.
