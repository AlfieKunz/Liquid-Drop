	\section{Reference Sheet}
	Force per unit area on fluid element: $\underline{F} = \int p \underline{n}\cdot d\underline{S} = \int_\Omega\nabla \cdot \mathbf{T} \ dV = \int_A \mathbf{T}\cdot \underline{n}\ dA$, where $\mathbf{T}\cdot \underline{n}$ is the stress on a surface normal to $\underline{n}$.
	
	$T_{ij} = -p\delta_{ij} + 2\mu e_{ij}$ is the Stress Tensor. Components $\{i,j\}$ is the stress along $i$ on the surface normal to $j$.
		
	Material Derivative: ${\frac{D}{Dt}} := {\frac{\partial }{\partial t} + (\underline{u}\cdot \nabla )}$.

	Navier Stokes Equations:
	\begin{subequations}
		\label{NavierStokes}
		\begin{align}
			\nabla \cdot \underline{u} = 0,\\
			\underbrace{\rho\left( \frac{\partial \underline{u}}{\partial t} + \overbrace{(\underline{u}\cdot \nabla) \underline{u}}^\text{inertial} \right)}_{ma} = \underbrace{\overbrace{\mu\nabla ^2 \underline{u}}^\text{viscous} - \nabla p + \rho \underline{g}}_{\sum \underline{F}}.
		\end{align}
	\end{subequations}
	
	Stokes' Equations:
	\begin{subequations}
		\label{Stokes}
		\begin{align}
			\nabla p = \mu \nabla ^2 \underline{u},\\
			\nabla \cdot \underline{u} = 0.
		\end{align}
	\end{subequations}
	
	Streamlines: $\frac{\partial \underline{x}(s)}{\partial s} = \underline{u}(\underline{x}(s),t)$.
	
	Reynolds Number: $\text{Re} = \frac{\rho UR}{\mu}$.
	
	Bernoulli's Equations:
	\begin{subequations}\label{Bernoulli}
		\begin{align}
			\tilde{B}(\underline{x}) &= \left( \frac{\rho}{2} |u(x)|^2 + p  + \rho gz \right),\quad&\text{(Steady, Inviscid. Constant on streamlines)} \label{Bernoulli1}\\
			\hat{B}(t) &= \left( {\rho \frac{\partial \phi}{\partial t}} + \frac{\rho}{2} |u(x,t)|^2 + p + \rho gz \right).\quad&\text{(Potential. Constant in space)} \label{Bernoulli2}
		\end{align}
	\end{subequations}
	
	Potential / Irrotational Flows: $\underline{u} = \nabla \phi(\underline{x},t) = \left( \frac{\partial \phi}{\partial r}, \frac{1}{r}\frac{\partial \phi}{\partial \theta} \right)$, with $\nabla ^2\phi = 0$.
	
	Incompressible Flows: $\underline{u} = \left( \frac{\partial \psi}{\partial y}, -\frac{\partial \psi}{\partial x} \right) = \left( \frac{1}{r}\frac{\partial \psi}{\partial \theta}, -\frac{\partial \psi}{\partial r} \right)$.
	
	These link via: $\{\partial _x\phi = \partial _y\psi,\ \partial _y\phi = -\partial _x\psi\}$.
	
	
	
	
	
	\section{Problem Overview}
	Bounces are modelled specifically via the thin air film that coats the drop - lubrication theory. The drop bounces on it's \textit{own air cushion}.
	Collisions with solids don't happen unless the solid is nano-smooth - small ridges puncture air film and reaches the liquid, sticking to it. Hence, drop-drop collisions (or symmetric to halve computation?).
	
	The closer the drop gets to the barrier (and the thinner the air film gets), the more pressure forms. With no attractive forces, this pressure builds up until it repels the drop. At this initial phase, we are attempting to recreate figure 3a from Hicks and Pervis.
	
	
	The drop is able to reach the barrier for two reasons: at small scales, the attract each other through van der Waals forces, and if the film height drops below the mean free path of gas molecules, pressure decreases, causing the gas molecules to slip along the boundary (Knudsen Gas-Kinetic Slip). If this happens, the the film ruptures and the drops merge.

	\subsection{Recap of Plane-Poiseuille Flow, and Lubrication Theory}
	The Plane-Poiseuille flow is the flow between two stationary plates of separation $h \sim \hat{\underline{z}}$.
	\begin{itemize}
		\item By the profile, we write $\underline{u} = u(y)\underline{\hat{x}}$. Via 1D nature, \textcolor[RGB]{255, 170, 0}{conservation of momentum} gives \underline{no inertial terms}, and:
		\begin{itemize}
			\item $\color{orange}x$: $\color{cyan}0 = -\frac{\partial p}{\partial x} + \mu \frac{\partial ^2u}{\partial y^2}$ \textit{(friction moves from boundary into the fluid)}, \quad $\color{orange}y$: $\color{lime} 0= -\frac{\partial p}{\partial y}$.
		\end{itemize}
		\item As $\color{lime}p(x)$, $u(y)$, we just \textcolor[RGB]{85, 255, 255}{integrate} to get $u(y) = \left( \frac{1}{2\mu} \frac{d p}{d x} \right) y^2 + Ay + B$.
		\item Using $u(y = \pm h/2) = 0$ (\underline{no-slip}, viscous), we get the \textbf{parabola} $\underline{u(y) = \frac{1}{2\mu} \frac{d p}{d x}\left( y^2 - \frac{h^2}{4} \right)}$.
	\end{itemize}
	
	\textbf{Lubrication Theory}: used as film radius $r \gg h$ height - very weak parabola. Hence, a fine assumption in this simple case is that the Poiseuille profile holds: $x,x^2 \to 0$ in the gas film, and we get a strip.

		\begin{align}
			\frac{1}{r} \frac{\partial (r u_r)}{\partial r} + \frac{1}{r} \frac{\partial u_\theta}{\partial \theta} + \frac{\partial u_z}{\partial z} &= 0, \label{NavierStokesND:continuity} \\

	\subsection{Lubrication Equation Derivation}
	By the axisymmetric conditions, $\partial / \partial \theta = 0$ and $u_\theta = 0$ \todo[color=cyan]{From "Stability of nonswirl axisymmetric solutions to the Navier-Stokes equations"}, and thus we get $\underline{u}(r,z)$. Hence, in the absence of external forces (including gravity), we get:
	
	\begin{subequations}\label{NavierStokesAXI}
		\begin{align}
			\frac{1}{r} \frac{\partial (r u_r)}{\partial r} + \frac{\partial u_z}{\partial z} &= 0, \label{NavierStokesAXI:continuity} \\
			\rho_g \left( \frac{\partial u_r}{\partial t} + u_r \frac{\partial u_r}{\partial r} + u_z \frac{\partial u_r}{\partial z} \right) &= -\frac{\partial p_g}{\partial r} + \mu_g \left[ \frac{\partial}{\partial r}\left( \frac{1}{r} \frac{\partial (r u_r)}{\partial r} \right) + \frac{\partial^2 u_r}{\partial z^2} \right], \label{NavierStokesAXI:r} \\
			\rho_g \left( \frac{\partial u_z}{\partial t} + u_r \frac{\partial u_z}{\partial r} + u_z \frac{\partial u_z}{\partial z} \right) &= -\frac{\partial p_g}{\partial z} + \mu_g \left[ \frac{1}{r}\frac{\partial}{\partial r}\left( r \frac{\partial u_z}{\partial r} \right) + \frac{\partial^2 u_z}{\partial z^2} \right]. \label{NavierStokesAXI:z}
		\end{align}
	\end{subequations}
	
	Rescaling the momentum equations under the regime of $h / R \ll 1$ (ie: locally flat), we conclude that, to leading order\todo[color=cyan]{cite Schlichting, 1968, Long-scale evolution of thin liquid films}:
	\begin{subequations}\label{BoundaryLayerScaling}
		\begin{align}
			\frac{\partial p_g}{\partial r} = \mu_g \frac{\partial^2 u_r}{\partial z^2},\label{BLScaling} \\
			\frac{\partial p_g}{\partial z} = 0.
		\end{align}
	\end{subequations}
	This directly concludes that $p_g$ is a function of $r$ and $t$ only! Using the boundary conditions of $\underline{u}(r,0) = \underline{u}(r,h(r,t)) = 0$, we can integrate \eqref{BLScaling} to get the Plane Poiseuille velocity profile \todo[color=cyan]{CITE: Long-scale evolution of thin liquid films}:
	\begin{equation}
		u_r(r,z,t) = \frac{z(z-h(r,t))}{2\mu_g} \frac{\partial p_g(r,t)}{\partial r},
	\end{equation}
	with velocity flux,
	\begin{equation}
		\label{Flux}
		Q(r,t) = \int^{h(r,t)}_0 u_r\, dz = -\frac{h^3}{12\mu_g} \frac{\partial p_g(r,t)}{\partial r}.
	\end{equation}
	To determine the evolution of the film thickness $h(r,t)$, we integrate the continuity equation \eqref{NavierStokesAXI:continuity} inside the entire $z$ domain \todo[color=cyan]{CITE NOTES 2 DERIVATION OF THE CLASSICAL REYNOLDS EQUATION FOR THIN FILM FLOWS} to get,
	\begin{equation}
		\label{ContInt}
		\int_0^{h(r,t)} \frac{1}{r} \frac{\partial (r u_r)}{\partial r}\ dz + \int_0^{h(r,t)} \frac{\partial u_z}{\partial z} \ dz = 0.
	\end{equation}
	Recalling Leibniz Integration Rule, and applying \eqref{BCKinetic}\todo[color=cyan]{CITE: BJ Hamrock, Fundamentals of Fluid Film Lubrication}, the first term in \eqref{ContInt} evaluates to
	\begin{equation}
		\label{ContInt1}
		\frac{1}{r} \frac{\partial}{\partial r}\left( r \int_0^h u_r\, dz \right) - u_r(r,h)\frac{\partial h}{\partial r} \ =\ \frac{1}{r} \frac{\partial (r Q)}{\partial r} - \left( u_z(r,h) - \frac{\partial h}{\partial t} \right),
	\end{equation}
	whilst the Fundamental Theorem of Calculus and boundary condition \eqref{BCNoSlip} can be used to evaluate the second term in \eqref{ContInt} as merely $u_z(r,h)$. Combining this with \eqref{ContInt1} in continuity equation \eqref{ContInt}, whilst subbing in \eqref{Flux} for $Q$, we retrieve the axisymmetric Reynolds lubrication equation \todo[color=cyan]{CITE: Modelling and closed loop control of near-field acoustically levitated objects, A rotational traveling wave based levitation device - Modelling, design, and control }:
	\begin{equation}
		\frac{\partial h(r,t)}{\partial t} = \frac{1}{r}\frac{\partial }{\partial r}\left( \frac{rh^3}{12\mu_g} \frac{\partial p_g(r,t)}{\partial r} \right).
	\end{equation}
	Scaling radial lengths by the drop radius $R$, film thickness by $H$, time by $R/U$, and pressures by the liquid inertial scale $\rho_l U^2$, this can be expressed in non-dimensional form as:
	\begin{equation}\label{LubricationND}
		\frac{\partial h(r,t)}{\partial t} = \frac{1}{r}\frac{\partial }{\partial r}\left( \frac{\text{St}}{12} rh^3 \frac{\partial p_g(r,t)}{\partial r} \right),
	\end{equation}
	where $\text{St} = \frac{\rho_l R U}{\mu_g}$ is the Stokes number.
	
	
	
	
	\subsection{Droplet Equations Formation}
	We must reset our assumptions of a thin film, and model $p_l(r,z)$ specifically as the liquid pressure evaluated at time $t$.
	We introduce the characteristic pressure length $\ell$ (the horizontal length that pressure makes a leading order contribution to the deformation of the drop), or rather the ratio $\delta := \ell/R$. Rescaling with this ($L \to \delta L$, $t \to \delta^2t$, $h \to \delta^2h$, $U \to U$ (and thus other components of drop velocity! Don't really care about oblique impacts unless they are VERY oblique - solutions match to leading order), $p \to p/\delta$), to leading-order as $z \to 0$ we lose the advective terms fo the material derivative $\frac{D}{Dt}$ \todo[color=cyan]{CITE: HP, Air cushioning with a lubrication/inviscid balance}. This has the following consequences on our formulation:
	\begin{itemize}
		\item \eqref{BCKinetic} becomes\todo[color=cyan]{CITE HP}
		\begin{equation}
			\label{KineticApprox}
			\frac{\partial h}{\partial t} = u_z(r, h).
		\end{equation}
		In other words, the interface $f(r,z)$ flattens entirely onto the $z=0$ plane.
		\item Recalling that we are in a high Reynolds Number approximation, the axisymmetric version of \eqref{NavierStokesND} becomes:
		\begin{equation}
			\label{NSHarmonic}
			\frac{\partial \underline{u}_l}{\partial t} = -\left( \frac{\partial p_l}{\partial r}, \frac{\partial p_l}{\partial z} \right).
		\end{equation}
	\end{itemize}
	(Again, good to stress that this is in the leading order approximation of $z \to 0$.)
	
	Considering initially the second of these points, taking the cylindrical divergence of \eqref{NSHarmonic}, and equating to \eqref{NavierStokesAXI:continuity} gives \todo[color=cyan]{CITE: HP, Air cushioning with a lubrication/inviscid balance},
	\begin{equation}
		\label{PressureLaplace}
		\nabla ^2 p_l = \frac{1}{r}\frac{\partial}{\partial r}\left(r \frac{\partial p_l}{\partial r}\right) + \frac{\partial^2 p_l}{\partial z^2} = 0.
	\end{equation}
	In other words, the pressure field is harmonic within the liquid \todo[color=cyan]{CITE C. Pozrikidis, Boundary Integral and Singularity Methods for Linearized Viscous Flow}. By the weak maximum principle for the Laplace Equation, we can therefore determine $p$ uniquely from the boundary conditions (for example, using the Dirichlet Principle). \todo[color=cyan]{cite: Evans, L. C. (2010). Partial Differential Equations}

	Considering now the first of these points, we are able to relate the downward acceleration of the interface to the pressure at the boundary by differentiating \eqref{KineticApprox} and using \eqref{NSHarmonic} to get:
	\begin{equation}
		\label{KineticODE}
		-\frac{\partial^2 h(r,t)}{\partial t^2} = \left.\frac{\partial p_l(r,z)}{\partial z}\right|_{z \to 0^+},
	\end{equation}		
	providing an ODE for the interface, which can be solved by the following recipe:
	\begin{itemize}
		\item Solve the Lubrication Equation for the thin film, given the previously produced value (or initial condition) for $\partial_t h(r,t)$, giving us an equation for $p_g(r,t)$.
		\item Use Bernoulli's equations to solve for the pressure boundary conditions of the liquid, via $p_l(r, h) = p_g(r, t)$ (exploiting the fact that $\partial _zp_g = 0$).
		\item Exploit the linearity of the harmonic liquid pressure to find an equation for the full liquid pressure.
		\item Use this pressure equation to thus solve the ODE \eqref{KineticODE}, giving us a new value for $h(r,t)$.
		\item Repeat!
	\end{itemize}
	
	We are left with the 3rd step of this list, which requires the introduction of Green's functions. Intuitively, these are the responses to Dirac Delta functions $\delta([r,z] - [r', z'])$, which in our axisymmetric coordinates can be thought of as an impulsive action in a ring in the $r-\theta$ plane, of radius $r'$ and height $z'$, felt at a response $(r,z)$. This response function, the \textbf{Green's Function} $G(r,z;\ r',z')$, allows us to build an entire solution for
	\begin{equation}
		\label{PressureDelta}
		p_l(r,z) = \iint \delta(r-r')\delta(z-z')p_l(r',z') \ dr'dz',
	\end{equation}
	by replacing the Dirac Deltas with the Poisson equation for the standard free-space Green's function. In axisymmetric cylindrical coordinates, this is given by the standard form \todo[color=cyan]{CITE Jackson, J. D. (1999) Classical Electrodynamics, C. Pozrikidis, Boundary Integral and Singularity Methods for Linearized Viscous Flow}
	\begin{equation}
		\label{GreenFreeSpace}
		-\nabla ^2G(r,z;\ r',z') = \delta([r,z] - [r', z']) := \frac{\delta(r-r')\delta(z-z')}{2\pi r'},
	\end{equation}
	where the factor $2\pi r'$ comes from symmetry of $\theta$ in cylindrical coordinates, and the axisymmetric differential operator\todo[color=cyan]{Jackson, J. D. (1999) Classical Electrodynamics},
	\begin{subequations}
		\label{NablaCyl}
		\begin{align}
			\nabla &:= \left( \frac{\partial }{\partial r'}\ ,\ \frac{\partial }{\partial z'} \right), \label{NablaCyl:1} \\
			\nabla^2 &:= \frac{1}{r'} \frac{\partial }{\partial r'} \left( r' \frac{\partial }{\partial r'} \right) + \frac{\partial ^2}{\partial z'^2}. \label{NablaCyl:2}
		\end{align}
	\end{subequations}
	
	Recalling that the eigenfunctions of the radial components of \eqref{NablaCyl:2} are the zeroth-order Bessel Functions $J_0(kr')$ \todo[color=cyan]{Jackson, J. D. (1999) Classical Electrodynamics} (where $k$ are eigenvector wavenumbers), which satisfy $\frac{1}{r'}\frac{d}{dr'}\big(r'\frac{d}{dr'}J_0(k r')\big) = -k^2 J_0(k r')$, we can write \eqref{GreenFreeSpace} in pure $(r,z)$ components by expressing it as a Hankel integral \todo[color=cyan]{CITE Jackson, J. D. (1999) Classical Electrodynamics, G. N. Watson, A Treatise on the Theory of Bessel Functions}:
	\begin{equation}
		\label{GreenHankel}
		G(r, z; r', z') = \frac{1}{4\pi} \int_0^\infty J_0(k r) J_0(k r') e^{-k|z-z'|} \, dk.
	\end{equation}
	Lipschitz–Hankel \todo[color=cyan]{CITE Jackson, J. D. (1999) Classical Electrodynamics, AND MORE PLEASE!!!!!} proved that this has a family of exact closed-form solutions \todo[color=cyan]{CITE C. Pozrikidis, Boundary Integral and Singularity Methods for Linearized Viscous Flow, Gradshteyn \& Ryzhik Table of Integrals, Series and Products 6.612}:
	\begin{equation}
		\label{GreenElliptic}
		G(r,z; r',z') = \frac{K(m)}{2\pi^2\sqrt{(r+r')^2+(z-z')^2}}, \qquad
		\text{where } m = \frac{4rr'}{(r+r')^2 + (z-z')^2}.
	\end{equation}
	In the above equation, $K(m)$ is the complete elliptical integral of the first kind, a function of the elliptic parameter $m$. The analysis to come will also make use of the second kind of said functions, $E(m)$. Both are defined as such: \todo[color=cyan]{CITE C. Pozrikidis, Boundary Integral and Singularity Methods for Linearized Viscous Flow, Resistance and mobility functions for the near-contact motion of permeable particles, Hicks Air trapping at impact of a rigid sphere onto a liquid, DLMF 19.4, Abramowitz, Stegun Handbook of Mathematical Function}
	\begin{subequations}
		\label{EllipticalIntegrals}
		\begin{align}
			K(m) &= \int^{\pi/2}_0 \frac{d\theta}{\sqrt{1-m\sin^2(\theta)}}\qquad = \int_0^1 \frac{dt}{\sqrt{(1-t^2)(1-mt^2)}}\ , \label{EllipticalIntegrals:K} \\
			E(m) &= \int^{\pi/2}_0 {\sqrt{1-m\sin^2(\theta)}}\ d\theta\quad = \int_0^1 \sqrt{\frac{1-mt^2}{1-t^2}} \ dt, \label{EllipticalIntegrals:E} \\
		&\qquad \qquad	K'(m) = \frac{E(m)}{2m(1-m)} - \frac{K(m)}{2m}. \label{EllipticalIntegrals:Derivative}
		\end{align}
	\end{subequations}
	"C. Pozrikidis, Boundary Integral and Singularity Methods for Linearized Viscous Flow" provides a fantastic derivation of these functions.
	
	At this point, we can intuitively convert the 2D problem into one defined on the droplet's boundary: a \textbf{boundary integral method} (BIM). This involves the use of Green's Second Identity\todo[color=cyan]{CITE: Partial Differential Equations: An Introduction, 2nd Edition}, which in our coordinates states,
	\begin{equation}
		\label{Green2}
		-p_l(r,z) = \iint_\Omega \left(p_l \nabla^2 G - G \nabla^2 p_l\right) 2\pi r' dr' dz' = \oint_{\partial\Omega} \left(p_l \frac{\partial G}{\partial n'} - G \frac{\partial p_l}{\partial n'}\right) 2\pi r' ds,
	\end{equation}
	where $\Omega$ is the drop's volume, $\partial \Omega$ is the interface's parametrisation, and $n'$ is the outward unit normal to $\partial \Omega$. Note that we have used \eqref{PressureDelta}, \eqref{GreenFreeSpace}, and \eqref{PressureLaplace} to equate the left-hand side to the liquid pressure. We can reduce $n' = -z'$, recalling that our leading order approximations of $z \to 0$ flattened the droplet interface onto the $z=0$ plane, and thus the outer `spherical' volume can be asymptotically reduced by taking the limit of $R = \sqrt{r'^2 + z'^2} \to \infty $\todo[color=cyan]{CITE C. Pozrikidis, Boundary Integral and Singularity Methods for Linearized Viscous Flow}. Thus, the integral parameter $ds = dr'$ (noting that we are implicitly integrating over $\theta$ to form a closed 'surface') with $z'=0 \in \partial \Omega$.
	
	However, note that one of the terms that remains stuck in the integral, $\frac{\partial p_l}{\partial z'}$, is redundant, as it the exact quantity we are wishing to solve for via \eqref{KineticODE}. However, we are able to add a term to both sides of the equation to subtract this variable \todo[color=cyan]{CITE C. Pozrikidis, Boundary Integral and Singularity Methods for Linearized Viscous Flow}! This process is called the \textbf{Method of Images}, and operates by subtracting a second Green's Function at $z = -z'$, effectively cancelling out the $z' = 0$ contributions. We get the new function:
	$$G(r,z; r',z') \to G_M(r,z;r',z') := G(r,z; r',z') - G(r,z; r',-z') = G(r,z; r',z') - G(r,-z; r',z'),$$
	where the final equality holds from the symmetry of Green's Functions (we are integrating over the primed, so this makes sense). By the linearity of Green's functions, we get $G_M = 0$ for $z=0$, and thus the $\frac{\partial p_l}{\partial z'}$ contribution in \eqref{Green2} (when replacing $G \to G_M$) drops out. Crucially, this substitution does \textbf{not} impact the left-hand side of the equation, as this second Green's Function does not exist inside of the drop's volume, $\Omega$!
	
	Note that in \eqref{Green2}, $n' = +z'$ for the first Green's function and $n' = z'$ for the second Green's function, as the flux felt from each Green's function heads in opposite directions. Hence, we get double the contribution for $\frac{\partial G}{\partial n'}$, and thus can modify the right-hand side of \eqref{Green2} to generate the full solution:
	\begin{equation}
		\label{PressureGreen}
		\begin{aligned}
			p_l(r, z) &= -\int_0^\infty p_l(r', 0) \left.\frac{\partial G_M}{\partial n'}\right|_{z'=0}\ 2\pi r' \, dr' \\
			&= -4\pi \int_0^\infty p_l(r', 0) \frac{\partial G(r,z;r',0)}{\partial z} \, r'\ dr' \\
			&= \int_0^\infty p_l(r', 0) \left[\int_0^\infty k J_0(k r) J_0(k r') e^{-k z} \, dk\right] r' \, dr'.
		\end{aligned}
	\end{equation}
	
	We now move to the penultimate step of our recipe: using \eqref{KineticODE} to form an ODE for $h(r,t)$ from \eqref{PressureGreen}. First however, we manipulate the form of \eqref{PressureGreen} slightly to better fit it numerically to \eqref{LubricationND}: as this equation directly produces $\frac{\partial p_g}{\partial r}$, and Bernoulli gives $p_g = p_l$ at the interface, it is much easier numerically to write $p_l(r, z)$ above as an $dr'$ integral of $\frac{\partial p_l}{\partial r'}$. Recalling the zeroth and first order Bessel function derivatives: \todo[color=cyan]{CITE Abramowitz, Stegun Handbook of Mathematical Functions}
	\begin{subequations}
		\label{BesselDerivatives}
		\begin{align}
			\frac{\partial }{\partial r} J_0(kr) &= -kJ_1(kr), \label{BesselDerivatives:1} \\
			krJ_0(kr) &= \frac{\partial }{\partial r} (rJ_1(kr)), \label{BesselDerivatives:2}
		\end{align}
	\end{subequations}
	we can apply integration by parts to the $dr'$ integral of \eqref{PressureGreen} to get
	\begin{equation}
		\label{PressureGreenIBP}
		\begin{aligned}
			p_l(r, z) &= \int^\infty _0 \left( \int^\infty _0 p_l(r',0) kr'J_0(kr') dr' \right) J_0(kr) e^{-kz} \ dk \\
			&= \int^\infty _0 \left( \left[ p_l(r',0)\ r'J_1(kr') \right]^\infty _0 - \int^\infty _0 \frac{\partial p_l(r',0)}{\partial r'} r'J_1(kr') dr' \right) J_0(kr) e^{-kz} \ dk \\
			&= -\int^\infty _0 \frac{\partial p_l(r',0)}{\partial r'} r' \int^\infty _0 J_0(kr)J_1(kr')e^{-kz}\ dk\ dr',
		\end{aligned}
	\end{equation}
	noting the far-field boundary condition $\lim_{r \to \infty }p_l(r ,0) = 0$ and Bessel function property $\lim{r \to 0}rJ_1(kr) = 0$ \todo[color=cyan]{CITE Jackson, J. D. (1999) Classical Electrodynamics, Abramowitz, Stegun Handbook of Mathematical Functions} to cancel the evaluated integral. Equating the Lipschitz–Hankel integral \eqref{PressureGreenIBP} with \eqref{KineticODE} in the limit of $z \to 0$, and recalling the derivative rule \eqref{BesselDerivatives:1}, 
	\begin{equation}
		\label{BoundaryIntegralAntiderivative}
		\begin{aligned}
			\frac{\partial^2 h(r, t)}{\partial t^2} = -\lim_{z \to 0} \frac{\partial p_l(r,z)}{\partial z} &= \lim_{z \to 0} \int^\infty _0 \frac{\partial p_l(r',0)}{\partial r'} r' \int^\infty _0 J_0(kr)\left[ \frac{\partial }{\partial r'}J_0(kr') \right]e^{-kz}\ dk\ dr' \\
			&= \int^\infty _0 \frac{\partial p_l(r',0)}{\partial r'} r' \left[ \frac{\partial }{\partial r'} \left( \frac{2K(m)}{\pi\sqrt{(r+r')^2 + z^2}} \right) \right]_{z=0}\ dr',
		\end{aligned}
	\end{equation}
	where the final step follows from the parametric form of Gradshteyn \& Ryzhik (2014), Equation 6.612.3 \todo[color=cyan]{Gradshteyn \& Ryzhik Table of Integrals, Series and Products 6.612}. Recalling the derivative rule of elliptical integrals \eqref{EllipticalIntegrals:Derivative}, with $\left. \frac{\partial m}{\partial r'} \right|_{z=z'=0} = \frac{4r(r-r')}{(r+r')^3}$, we finally produce the analytic ODE for the interface $h(r,t)$:
	\begin{equation}
		\label{BoundaryIntegral}
		\begin{aligned}
			\frac{\partial^2 h(r, t)}{\partial t^2}	&= \int^\infty _0 \frac{\partial p_l(r',0)}{\partial r'} r' \left[ \frac{1}{\pi r'} \left( \frac{E(m)}{r-r'} - \frac{K(m)}{r+r'} \right) \right]\ dr' \\
			&= \frac{1}{\pi} \, \text{PV} \int_0^\infty \frac{\partial p_l(r',0)}{\partial r'} \, \left[\frac{E(m)}{r - r'} - \frac{K(m)}{r + r'}\right] dr', \quad \text{where } m = \frac{4rr'}{(r+r')^2}.
		\end{aligned}
	\end{equation}
	
	Note also that the equation for the elliptic parameter $m$ differs slightly compared with \eqref{GreenElliptic} - this is because our integral interface is performed over the $z'=0$ plane, in the limit of $z \to 0$. Further, as $m$ is the square of the geometric mean to the arithmetic mean of $r$ and $r'$, $\sqrt{m} \le 1$, and so the elliptical functions are well-defined \todo[color=cyan]{CITE air trapping at impact hicks}.


	\subsubsection{Desingularisation} \label{sec:CauchyPV}
	We have introduced the notation $\text{PV}$ in \eqref{BoundaryIntegral} above to mean the `Cauchy Principal Value', which applies only to the simple pole produced by the $E(m)$ integral, and intuitively means to exclude two small intervals on either side of the singularity $r = r'$ (for which $E(m) = 1$, producing a simple pole), and taking the limit as the size of these intervals tends to zero in a simultaneous manner \todo[color=cyan]{CITE C. Pozrikidis, Boundary Integral and Singularity Methods for Linearized Viscous Flow}. As \todo[color=cyan]{CITE Abramowitz \& Stegun With Formulas, Graphs and mathematical tables}
	\begin{equation}
		\label{KExpansion}
		K(m) \sim \ln \left( \frac{4(r+r')}{|r-r'|} \right), \quad \text{as } m \to 1^-,
	\end{equation}
	$r=r'$ is also a singularity for $K(m)$, but is an integrable logarithmic singularity, rather than a pole (ie: it has finite area, but still diverges when evaluated point-wise). Whilst not strictly necessary for computation, desingularising the simple pole introduced by $E(m)$ eliminates an $O(1/\Delta r)$ sources of error, and further handling of the logarithmic singularity introduced by $K(m)$ eliminates a further $O(1/\ln \Delta r)$ source of error. Note that in this numerical discussion, we therefore replace the semi-improper integral with one with an upper bound $R$: not only does this aid the conceptual understanding of the numerical methods, but also allows the Cauchy Principal Value area of the singularity $1/r$ to be finite.
	
	To handle the $E(m)$ singularity, we use a method called `singularity subtraction' \todo[color=cyan]{CITE HP, OTHER PLEASE!}: we aim to replace the $\frac{E(m)}{r-r'}$ term in the kernel of \eqref{BoundaryLayerScaling} with $\frac{E(m) - 1}{r-r'}$, which is of order $\sim O(|r-r'|\ln|r-r'|)$\todo[color=cyan]{CITE DLMF 19.12 Abramowitz \& Stegun With Formulas, Graphs and mathematical tables } and is therefore numerically stable. Adding and subtracting the pure simple pole $\frac{\partial _rp_l(r,0)}{r-r'}$, which has an exact solution as given by I.M. Gel’fand and G.E. Shilov\todo[color=cyan]{CITE Gel’fand, I. M., \& Shilov, G. E. (1964). Generalized Functions}, therefore gives:
	\begin{equation}
		\begin{aligned}
			\text{PV}\int^R _0 \frac{\partial p_l(r')}{\partial r'}r'\ \frac{E(m)}{r-r'}\ dr' &= \int^R _0 \frac{\partial _{r'}p_l(r')r'E(m) - \partial _{r}p_l(r)r}{r-r'}\ dr' + \frac{\partial p_l(r)}{\partial r}r\ \text{PV}\int^R _0 \frac{1}{r-r'}\ dr' \\
			&= \int^R _0 \frac{\partial p_l(r')}{\partial r'}r'\ \frac{E(m)}{r-r'}\ dr'\ +\  \frac{\partial p_l(r)}{\partial r}r \ln\left( \frac{r}{R - r} \right).
		\end{aligned}
	\end{equation}
	Considering now the $K(m)$ singularity, we use Stroud \& Secrest (1966) (Log-Gaussian) Quadrature \todo[color=cyan]{CITE}. Rearranging \eqref{KExpansion} to get
	\begin{equation}
		\frac{K(m)}{r + r'} \approx \frac{\ln(4(r + r'))}{r + r'} - \frac{\ln|r-r'|}{r + r'},
	\end{equation}
	and thus by setting $r+r' = 2r$, to leading order we get that $\frac{K(m)}{r + r'} \sim \frac{\ln|r-r'|}{2r}$ as $m \to 1^-$! Hence, we add and subtract this quantity to get
	$$\frac{K(m)}{r + r'} = {\left[\frac{K(m)}{r + r'}   \frac{\ln|r-r'|}{2r} \right]} - \frac{\ln|r-r'|}{2r},$$
	the sum of a smooth, integrable function with a pure logarithmic spike, which has an exact closed form as given by the Stroud \& Secrest Log-Gaussian Quadrature:
	$$\int_{r}^{r + \Delta r/2} \partial _{r'}p_l(r')r'\ln\vert{}r - r'\vert{}\,dr' = \left[ \ln\left(\frac{\Delta r}{2}\right)\underbrace{\int_0^1 \tilde{\partial _{r'}p_l(r')r'}(\xi)\,d\xi}_{\text{Standard Gauss}} - \underbrace{\int_0^1 \tilde{\partial _{r'}p_l(r')r'}(\xi)\ln\left(\frac{1}{\xi}\right)d\xi}_{\text{Stroud \& Secrest Table}} \right] \frac{\Delta r}{2}.$$

	\subsection{Numerical Evaluation of the Boundary Integral \eqref{BoundaryIntegral}}
	In the derivation as outlined in \ref{sec:CauchyPV}, we use the Cauchy Principal Value to evaluate the elliptical integral singularities. This is an exact expression, but as FVM methods are of order $O(\Delta r^2)$, we can make a much cheaper approximation and still generate the same results to leading order, as long as all the integrand components are of order $O(\Delta r^2)$! Specifically, we will assume that the function $f(r) := r \frac{\partial p_l(r,0)}{\partial r}$ varies \underline{linearly} across a cell of size $\Delta r_j$ (see the verification below), where
	\begin{subequations}
		\label{fFVMDefinition}
		\begin{align}
			& \qquad f(r') = f(r_j) + f'(r_j)(r' - r_j), \label{fFVMDefinition:1} \\[6pt]
			f(r_j) = &\frac{f_{j+1/2} + f_{j-1/2}}{2}, \qquad f'(r_j) = \frac{f_{j+1/2} - f_{j-1/2}}{\Delta r_j},  \label{fFVMDefinition:2}
		\end{align}
	\end{subequations}
	(in order to define $p$ at the centre of each cell). Thus, continuing from the quadrature form of equation \eqref{BoundaryIntegralAntiderivative} by piece-wise integrating over each cell $(r_{j-1/2}, r_{j+1/2})$ and Taylor expanding in this small region, we get:
	\begin{equation}
		\label{BIAQuad}
		\begin{aligned}
			\frac{\partial^2 h(r_i,t)}{\partial t^2} &\approx \sum^N_{j = 1} \int^{r_{j+1/2}}_{r_{j-1/2}} f(r') \frac{\partial \mathcal{K}(r_i, r')}{\partial r'}\ dr'\quad \left( \mathcal{K}(r, r') := \frac{2K(m)}{\pi(r+r')},\ m(r,r') = \frac{4rr'}{(r+r')^2} \right) \\
			&\approx \sum^N_{j = 1} f(r_j) \underbrace{\int^{r_{j+1/2}}_{r_{j-1/2}} \frac{\partial \mathcal{K}(r_i, r')}{\partial r'}\ dr'}_{\mathbf{K}^{(0)}_{ij}} + f'(r_j) \underbrace{\int^{r_{j+1/2}}_{r_{j-1/2}}  (r'-r_j)\frac{\partial \mathcal{K}(r_i, r')}{\partial r'}\ dr'}_{\mathbf{K}^{(1)}_{ij}}.
		\end{aligned}
	\end{equation}
	where $r_{N+1/2} = R$ and, as mentioned in Section \ref{sec:CauchyPV}, $R \gg 1$ is the upper bound of our simulation, chosen large enough to approximate the far-field boundary condition $\lim_{r \to \infty }p_l(r,z) = 0$. As previously covered, the kernel $\mathcal{K}(r_i,r')$ contains an integrable logarithmic singularity at $r' = r_i$ (only in the summation term $i = j$), and thus we can cleanly apply the Fundamental Theorem of Calculus to the $\frac{\partial \mathcal{K}}{\partial r'}$ term.
	
	In the last step of \eqref{BIAQuad}, we have also performed Taylor expansion over $f(r)$, however it is important to verify that this is sufficiently smooth, such that neglecting higher-order derivatives still gives a solid approximation, thus ensuring that we don't introduce a significant bottleneck in the second-order accuracy of Finite Volume Method discretisation. In particular, we show that the linear approximation of $f$ is bounded by the quadratic Taylor remainder $\frac{1}{2}f''(r)\Delta r^2$ at $O(\Delta r^2)$.
	
	We take $\frac{\partial p_g(r,t)}{\partial r} r$ from the lubrication equation \eqref{LubricationND}, and use Bernoulli to equate this to $f(r)$. For the initial interface $h = y + r^2/2$, where $y = h_0(0,0) -Ut$ is the gap between the interface and the solid, we define the characteristic length $\ell = \sqrt{2y}$ (the radius for which $h = 2y$, or rather the characteristic pressure width).
	Rearranging \eqref{LubricationND} gives $f(r)=-6r^2/(\text{St}\,h^3)$. Rescaling via the variable $\xi = r/\ell$ gives $f(\xi)=-\frac{12}{\text{St}\,y^2}\,\frac{\xi^2}{(1+\xi^2)^3}$, a smooth, single-humped function $\approx\xi^2$ near the axis, peaking at $\xi=1/\sqrt2$, and decaying like $\xi^{-4}$.
	
	The variation of $f$ across one cell can be written $\frac{|f(r+\Delta r)-f(r)|}{|f|_{\max}}$, whilst the remainder after linear approximation can be written $\frac{|f(r') - [f(r) + f'(r)(r'-r)]|}{|f|_{\max}}$.	Writing $\epsilon=\Delta r/\ell$, by the chain rule on $\xi$, we can therefore bound above the former of these by $\frac{|f'|_\text{max}}{|f|_\text{max}}\Delta r = c_1\epsilon$ (where, analytically, $c_1 \approx 2.36$), and the latter of these by $\frac{1}{2}\frac{|f''|_\text{max}}{|f|_\text{max}}\Delta r^2 = c_2\epsilon^2$ (where, analytically, $c_2 = 6.75$). Hence, provided $\epsilon \ll 1$, our linear approximation of $f$ matches the Finite Volume Method truncation error.	

	\todo[inline, color=green]{Note that this validation only works in the case of a rigid interface - naturally the film will shape as the droplet falls, preventing $\ell$ from reaching 0. Something something scaling, $\ell$ has a natural lower bound of $\mathcal{O}(\text{St}^{-1/3})$, or the neck length $\ell_\text{neck}$, whichever is smaller. Hence, as long as we enforce $\Delta r \ll \text{St}^{-1/3}$ (in our simulation, we use $\Delta r = 0.05 \min\{\text{ahh}\}$), we get the behaviour as above.}
	
	Despite this global $\mathcal{O}(\Delta r)$ behaviour, the above analysis reveals that locally in the inner-most cells (as $\xi \to 0$), though $f(\xi)$ is flat, $f''(\xi) \to 2$ (much like the standard 2nd-order polynomial)! Ie: to leading order $f(0)$ is approximated by a \underline{quadratic} piece-wise function. To avoid excessive calculations in this local region, however, all that is needed is to push this $f''(\xi)$ information into its gradient, which is captured by the locally affine representation of some function $g(r)$ (which has the same global behaviour of $f$). An easy way to handle this is to enforce $g(r)$ odd, for which a suitable form $f(r) = r g(r)$ emerges. Trivially, $g(r)$ must therefore be equal to $\frac{\partial p_l(r,0)}{\partial r}$, and this is the function we assume is locally affine:
	\begin{equation}
		\label{fExpansion}
		\begin{aligned}
			f(r') &\approx f(r_j) + \left[ g(r_j) +  g'(r_j)r_j \right](r'-r_j) \\
			&= r'\left[ g(r_j) + g'(r_j) (r' - r_j) \right].
		\end{aligned}
	\end{equation}
	Note that $g(r) \approx f(r)$ in the far-field approach, so we get the same $\mathcal{O}(\Delta r)$ behaviour globally - we do push lots of information to $g'''(r)$ as a result of this odd behaviour, however this is well-bounded, missed by our half-index $r_{i \pm 1/2}$, and of order $\mathcal{O}(\Delta r^2)$.
	
	Adding and subtracting the term $r_j$ to the preceding $r'$ term in \eqref{fExpansion} gives,
	\begin{equation}
		\label{fAddSubtract}
		\begin{aligned}
			f(r') &= \left[r_j + (r' - r_j)\right]\left[g(r_j) + g'(r_j)(r' - r_j)\right] \\
			&= r_j g(r_j) + \left[g(r_j) + r_j g'(r_j)\right](r' - r_j) + g'(r_j)(r' - r_j)^2 \\
			&= f(r_j) + f'(r_j) (r'-r_j) + g'(r_j)(r'-r_j)^2,
		\end{aligned}
	\end{equation}
	where each coefficient $\{f(r_j), f'(r_j), g'(r_j)\}$ is constant across cell $j$. This allows us to both rewrite equations \eqref{fFVMDefinition:2} in the case of $g$ locally affine rather than $f$ locally affine, as
	\begin{equation}
		\label{gFVMDefinition}
		f(r_j) = \frac{f_{j+1/2} + f_{j-1/2}}{2} - g'(r_j)\left(\frac{\Delta r_j}{2}\right)^2, \qquad f'(r_j) = \frac{f_{j+1/2} - f_{j-1/2}}{\Delta r_j},
	\end{equation}
	and replace each term of the summation in \eqref{BIAQuad} with the updated:
	\begin{equation}
		\label{BIAQuadg}
		\int^{r_{j+1/2}}_{r_{j-1/2}} f(r') \frac{\partial \mathcal{K}(r_i, r')}{\partial r'}\ dr' = f(r_j)\mathbf{K}^{(0)}_{ij} + f'(r_j)\mathbf{K}^{(1)}_{ij} + g'(r_j)\underbrace{\int^{r_{j+1/2}}_{r_{j-1/2}} (r'-r_j)^2 \frac{\partial \mathcal{K}(r_i, r')}{\partial r'}\ dr'}_{\mathbf{K}^{(2)}_{ij}}.
	\end{equation}
	In other words, the problem reduces to the case of the locally affine $f(r)$ approximation, plus the extra term $g'(r_j)\mathbf{K}^{(2)}_{ij}$. Applying the Fundamental Theorem of Calculus and integration by parts to $\mathbf{K}^{\{0, 1, 2\}}_{ij}$ (using equations \eqref{BIAQuad} and \eqref{BIAQuadg},
	\begin{subequations}
		\label{KIntegralByParts}
		\begin{align}	
			\mathbf{K}^{(0)}_{ij} &= \mathcal{K}(r_i, r_{j+1/2}) - \mathcal{K}(r_i, r_{j-1/2}) \\[-10pt]
			\mathbf{K}^{(1)}_{ij} &= \frac{\Delta r_j}{2}\left[\mathcal{K}(r_i,r_{j+1/2}) + \mathcal{K}(r_i,r_{j-1/2})\right] - \overbrace{\int^{r_{j+1/2}}_{r_{j-1/2}} \mathcal{K}(r_i,r')\ dr'}^{\mathbf{K}^P_{ij}}, \label{KIntegralByParts:1} \\
			\mathbf{K}^{(2)}_{ij} &= \left(\frac{\Delta r_j}{2}\right)^2\left[\mathcal{K}(r_i, r_{j+1/2}) - \mathcal{K}(r_i, r_{j-1/2})\right] - 2 \underbrace{\int_{r_{j-1/2}}^{r_{j+1/2}} (r' - r_j)\mathcal{K}(r_i, r')\, dr'}_{\mathbf{K}^{S}_{ij}}. \label{KIntegralByParts:2}
		\end{align}
	\end{subequations}
	First considering the simpler case of $f$ affine (thus neglecting the component $\mathbf{K}^{(2)}_{ij}$) to produce a leading-order solution to \eqref{BIAQuad}, applying formula \eqref{fFVMDefinition:2} gives:
	\begin{equation}
		\label{BIAQuad2}
		\begin{aligned}
			\frac{\partial^2 h(r_i,t)}{\partial t^2} &\approx \sum^N_{j = 1} f(r_{j+1/2})\mathcal{K}(r_i, r_{j+1/2}) - f(r_{j-1/2})\mathcal{K}(r_i, r_{j-1/2}) - f'(r_j)\mathbf{K}^P_{ij} \\
			&= f(R)\mathcal{K}(r_i, R) - f(0)\mathcal{K}(r_i, 0) - \sum^N_{j = 1} f'(r_j)\mathbf{K}^P_{ij} \quad \text{(by telescoping)} \\
			&\approx -\sum^N_{j = 1} f'(r_j) \mathbf{K}^P_{ij}\quad \text{(by boundary conditions)}.
		\end{aligned}
	\end{equation}
	
	Finally, by considering $g(r)$ locally affine rather than $f(r)$, and thus adding in the component $g'(r_j)\mathbf{K}^{(2)}_{ij}$ from \eqref{KIntegralByParts:2}, we substitute \eqref{KIntegralByParts} into \eqref{BIAQuadg}. Note that in this case, we use the approximation method \eqref{gFVMDefinition} rather than \eqref{fFVMDefinition:2}, to include this locally affine behaviour. By telescoping and applying boundary conditions as before, we generate the full numerical solution to the interface:
	\begin{equation}
		\label{BIAQuadFull}
		\begin{aligned}
			\frac{\partial^2 h(r_i,t)}{\partial t^2} &\approx \sum^N_{j = 1} f(r_j)\mathbf{K}^{(0)}_{ij} + f'(r_j)\mathbf{K}^{(1)}_{ij} + g'(r_j)\mathbf{K}^{(2)}_{ij} \\
			&= \sum^N_{j = 1} \underbrace{\left[f(r_j) + f'(r_j)\frac{\Delta r_j}{2} + g'(r_j)\left(\frac{\Delta r_j}{2}\right)^2\right]}_{f(r_{j+1/2})}\mathcal{K}(r_i, r_{j+1/2}) \\
			&\qquad\quad - \underbrace{\left[f(r_j) - f'(r_j)\frac{\Delta r_j}{2} + g'(r_j)\left(\frac{\Delta r_j}{2}\right)^2\right]}_{f(r_{j-1/2})}\mathcal{K}(r_i, r_{j-1/2}) - f'(r_j)\mathbf{K}^P_{ij} - 2g'(r_j)\mathbf{K}^S_{ij} \\
			&\approx \underline{-\sum^N_{j = 1} f'(r_j)\mathbf{K}^P_{ij} + 2g'(r_j)\mathbf{K}^S_{ij}} \quad \text{(by telescoping and boundary conditions)} \\[6pt]
			&\approx \sum_{j=1}^N \left[ \frac{r_{j-1/2}\ g(r_{j-1/2}) - r_{j+1/2}\ g(r_{j+1/2})}{\Delta r_j} \mathbf{K}^P_{ij} + 2\,\frac{g(r_{j-1/2}) - g(r_{j+1/2})}{\Delta r_j} \mathbf{K}^S_{ij} \right].
		\end{aligned}
	\end{equation}
	
	
	The entire problem thus reduces to the computing elements of the matrices $\mathbf{K}^R_{ij}$ and $\mathbf{K}^{(1)}_{ij}$. To reduce floating point errors near the singularity (where $m \to 1$), we use the Landen Transform \todo[color=cyan]{CITE DLMF 19.8.12, Landen} \texttt{scipy.special.ellipkm1$\left(m_1 := 1 - m = \frac{\min(r_i, r')^2}{\max(r_i, r')^2}\right)$}. For the off-diagonal terms, note that $\mathcal{K}(r_i, r')$ is completely smooth for $r'$ inside cell $j$, and thus integrating via a standard 20-point Gauss–Legendre quadrature \texttt{np.polynomial.legendre.leggauss(20)} suffices to balance accuracy with ease of computation. For the leading diagonal, we take advantage of the \texttt{QUADPACK} feature of \texttt{scipy.integrate.quad}, which uses Gauss-Kronrod rules with Wynn epsilon-extrapolation to handle the $K(m)$ logarithmic singularity. Our simulation uses the parameters \texttt{(points=[ri], epsabs=0, epsrel=1e-13)}.

	\subsubsection{Failed Asymptotic Expansions} \label{sec:FailedAsymptotics}
	
	We return back to equation \eqref{BIAQuad}. Applying integration by parts for the case of $i \neq j$, recalling that $\frac{\partial \mathcal{K}}{\partial r'}$ is completely smooth in this case and thus we only need to be consider $\mathbf{K}^{(1)}_{ij}$ at constant order to match the asymptotic expansion of $f'(r_j)$ locally constant:
	\begin{equation}
		\label{BIANonSingular}
		\begin{aligned}
			\mathbf{K}^{(1)}_{ij} &= \left[(r' - r_j)\mathcal{K}(r_i,r')\right]^{r_{j+1/2}}_{r_{j-1/2}} - \overbrace{\int^{r_{j+1/2}}_{r_{j-1/2}} \mathcal{K}(r_i,r')\ dr'}^\star \\
			&\approx \frac{\Delta r_j}{2}\left[\mathcal{K}(r_i,r_{j+1/2}) + \mathcal{K}(r_i,r_{j-1/2})\right] - \left[ \frac{\mathcal{K}(r_i,r_{j+1/2}) + \mathcal{K}(r_i,r_{j-1/2})}{2} \right]\Delta r_j \\
			&= 0 \quad (\text{provided } i \neq j).
		\end{aligned}
	\end{equation}
	\todo[inline]{Okay, so apparently this isn't the case - this is kinda the case for $i$ pretty far from $j$, but for direct neighbours we lose around 10\%! Basically, we can't chuck all information in the leading diagonal!}

	Writing $r' = r_i + \epsilon$, where $\epsilon \ll r_i$,

			&= \frac{\Delta r}{\pi r_i}\ln\left(\frac{16r_i}{\Delta r}\right) - \frac{2}{\pi r_i}\int^{\Delta r/2}_0 \ln\left(\frac{8r_i}{\epsilon}\right)\ d\epsilon + \mathcal{O}\left(\frac{\Delta r^3}{r_i^3}\ln\frac{r_i}{\Delta r}\right) \\
